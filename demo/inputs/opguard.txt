859:     def initialize_gradient_partition(self, i, param_group, partition_id):

861:         def set_key_value_list(dictionary, key, value):
862:             if key in dictionary:
863:                 dictionary[key].append(value)
864:             else:
865:                 dictionary[key] = [value]
866:
867:         def increment_value(dictionary, key):
868:             if key in dictionary:
869:                 dictionary[key] += 1
870:             else:
871:                 dictionary[key] = 1

873:         partition_size = self.partition_size[i]
874:
875:         start_index = partition_size * partition_id
876:         end_index = partition_size * (partition_id + 1)

879:         first_offset = 0

881:         for param in param_group:
882:

885:
886:             if start_index <= current_index < end_index:
887:                 set_key_value_list(self.param_to_partition_ids[i], param_id, partition_id)
888:                 increment_value(self.total_grads_in_partition[i], partition_id)
